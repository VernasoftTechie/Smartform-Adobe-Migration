*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      <-- TOTAL_PAYMENT
*&---------------------------------------------------------------------*
*&---------------------------------------------------------------------*
*& Form debit_credit
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> TOTAL_PAYMENT
*&      <-- LS_AMOUNT_PAYMENTS_RECEIVED
*&---------------------------------------------------------------------*
FORM debit_credit  USING    p_total_payment
                   CHANGING p_ls_amount.
  DATA lv_value TYPE zamt_loc.
  IF p_total_payment LT 0.
    lv_value = p_total_payment.
    lv_value = lv_value * -1.
    WRITE lv_value TO p_ls_amount.
    CONDENSE p_ls_amount NO-GAPS.
    CONCATENATE p_ls_amount '(Cr)' INTO p_ls_amount .
  ELSEIF p_total_payment GT 0.
    WRITE p_total_payment TO p_ls_amount.
    CONDENSE p_ls_amount NO-GAPS.
    CONCATENATE p_ls_amount '(Dr)' INTO p_ls_amount .
  ELSE.
    WRITE p_total_payment TO p_ls_amount.
    CONDENSE p_ls_amount NO-GAPS.
  ENDIF.
ENDFORM.
