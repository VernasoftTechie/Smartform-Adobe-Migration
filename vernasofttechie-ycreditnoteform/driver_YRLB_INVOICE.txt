*&---------------------------------------------------------------------*
FORM send_email .

  DATA : lo_sender      TYPE REF TO cl_cam_address_bcs,
         lo_recipient   TYPE REF TO if_recipient_bcs,
         lv_sent_to_all TYPE os_boolean.
  CONSTANTS lc_param TYPE sychar30 VALUE 'ZDMS_SENDER_ADDRESS'.

  SELECT SINGLE value1  FROM zab_dcp_param INTO ( gv_sender_email )
                                           WHERE param1 = lc_param.
  IF sy-subrc EQ 0.
    "Email FROM..
    lo_sender = cl_cam_address_bcs=>create_internet_address( gv_sender_email ).

    "Add sender to send request
    TRY.
        CALL METHOD go_send_request->set_sender
          EXPORTING
            i_sender = lo_sender.
      CATCH cx_address_bcs  INTO DATA(lo_address_bcs).
    ENDTRY.
  ENDIF.

**    Recipient mail ID
  SELECT SINGLE adrnr     FROM kna1 INTO @DATA(lv_adrnr) WHERE kunnr      = @gv_kunnr.
  IF sy-subrc EQ 0.
    SELECT SINGLE smtp_addr FROM adr6 INTO @DATA(lv_recep) WHERE addrnumber = @lv_adrnr.
  ENDIF.

  IF lv_recep IS NOT INITIAL.
    TRY.
        lo_recipient = cl_cam_address_bcs=>create_internet_address( lv_recep ).
      CATCH cx_address_bcs  INTO lo_address_bcs.
    ENDTRY.
    TRY.
        go_send_request->add_recipient( i_recipient = lo_recipient
                                       i_express = 'X'    ).
      CATCH cx_send_req_bcs INTO DATA(lo_send_req_bcs).
    ENDTRY.
    TRY.
        CALL METHOD go_send_request->send(
          EXPORTING
            i_with_error_screen = 'X'
          RECEIVING
            result              = lv_sent_to_all ).

      CATCH cx_send_req_bcs INTO lo_send_req_bcs.
    ENDTRY.
    IF lv_sent_to_all = 'X'.
      COMMIT WORK.
      gv_ind = 1.                        "Sent successfully
    ENDIF.

  ELSE.
    gv_ind = 2.                          "Recipient mail ID is not available
  ENDIF.



ENDFORM.
